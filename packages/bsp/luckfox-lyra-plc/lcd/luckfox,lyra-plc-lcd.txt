# panel-mipi-dbi init sequence for the Lyra PLC's 1.9" 170x320 ST7789V glass,
# in the mipi-dbi-cmd text format; the board hook compiles it to
# /lib/firmware/luckfox,lyra-plc-lcd.bin. The values are the ones the staging
# fb_st7789v driver sends for this (HSD20 IPS) glass.

command 0x11			# exit sleep mode
delay 120
command 0x3a 0x05		# 16 bits per pixel (RGB565)
command 0xb2 0x05 0x05 0x00 0x33 0x33	# PORCTRL: porch setting
command 0xb7 0x75		# GCTRL: gate voltages
command 0xc2 0x01 0xff		# VDVVRHEN: VDV and VRH from these commands
command 0xc3 0x13		# VRHS
command 0xc4 0x20		# VDVS: VDV 0 V
command 0xbb 0x22		# VCOMS
command 0xc5 0x20		# VCMOFSET: VCOM offset 0 V
command 0xd0 0xa4 0xa1		# PWCTRL1: AVDD 6.8 V, AVCL -4.8 V, VDS 2.3 V
command 0x36 0x60		# address mode MV | MX: landscape, as fbtft's rotate = 270
command 0xe0 0xd0 0x05 0x0a 0x09 0x08 0x05 0x2e 0x44 0x45 0x0f 0x17 0x16 0x2b 0x33	# PVGAMCTRL
command 0xe1 0xd0 0x05 0x0a 0x09 0x08 0x05 0x2e 0x43 0x45 0x0f 0x16 0x16 0x2b 0x33	# NVGAMCTRL
command 0x21			# display inversion on (IPS)
command 0x29			# display on
